import { useState } from "react";
import { Card, CardContent } from "@/components/ui/card";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";

export default function Home() {
  const [situation, setSituation] = useState("");
  const [analysis, setAnalysis] = useState("");

  const handleAnalyze = () => {
    // כאן יהיה חיבור ל-AI בהמשך
    setAnalysis("מנתח... (כאן תוצג התובנה מהבינה המלאכותית)");
  };

  return (
    <div className="p-4 space-y-4 max-w-md mx-auto">
      <h1 className="text-2xl font-bold text-center">יומן תובנות יומי</h1>

      <Card>
        <CardContent className="space-y-2 pt-4">
          <Textarea
            placeholder="כתוב כאן את הסיטואציה שחווית..."
            value={situation}
            onChange={(e) => setSituation(e.target.value)}
          />
          <Button onClick={handleAnalyze} className="w-full">
            נתח לי את זה
          </Button>
        </CardContent>
      </Card>

      {analysis && (
        <Card>
          <CardContent className="pt-4">
            <p className="font-semibold">תובנות:</p>
            <p>{analysis}</p>
          </CardContent>
        </Card>
      )}
    </div>
  );
}
